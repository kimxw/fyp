def bubble_sort(nums):
    """
    Sorts a list of numbers using the bubble sort algorithm.

    Args:
        nums (list): A list of numbers to sort.

    Returns:
        list: The sorted list of numbers.
    """

    # Get the length of the list
    n = len(nums)

    # Perform n-1 passes through the list
    for i in range(n-1):
        
        # Initialize a flag to track if any swaps were made
        swapped = False
        
        # Iterate through the list from the first element to the (n-i-1)th element
        for j in range(n-i-1):
            
            # If the current element is greater than the next element, swap them
            if nums[j] > nums[j+1]:
                nums[j], nums[j+1] = nums[j+1], nums[j]
                swapped = True
        
        # If no swaps were made in this pass, the list is already sorted
        if not swapped:
            break

    return nums
