def bubble_sort(num_list):
    """
    Sorts a list of numbers in ascending order using bubble sort.

    Args:
        num_list (list): A list of numbers to be sorted.

    Returns:
        list: A sorted list of numbers.
    """
    # Create a copy of the input list to avoid modifying it in place
    num_list = num_list.copy()

    # Get the length of the list
    n = len(num_list)

    # Iterate over the list n-1 times
    for i in range(n-1):
        # Initialize a flag to track if any swaps were made in the current pass
        swapped = False

        # Iterate over the list from the first element to the (n-i-1)th element
        for j in range(n-i-1):
            # If the current element is greater than the next element, swap them
            if num_list[j] > num_list[j+1]:
                # Swap the elements
                num_list[j], num_list[j+1] = num_list[j+1], num_list[j]
                # Set the flag to True to indicate that a swap was made
                swapped = True
