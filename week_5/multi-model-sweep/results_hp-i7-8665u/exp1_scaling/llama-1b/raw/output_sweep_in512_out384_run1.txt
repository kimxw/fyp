def bubble_sort(numbers):
    """
    Sorts a list of numbers using the bubble sort algorithm.

    Args:
        numbers (list): The list of numbers to sort.

    Returns:
        list: The sorted list of numbers.
    """
    # Get the length of the list
    n = len(numbers)

    # Repeat the process until no more swaps are needed
    for i in range(n):
        # Initialize a flag to check if any swaps were made in the current iteration
        swapped = False

        # Iterate through the list from the first element to the (n-i-1)th element
        for j in range(n - i - 1):
            # Compare each pair of adjacent elements
            if numbers[j] > numbers[j + 1]:
                # Swap the elements if they are in the wrong order
                numbers[j], numbers[j + 1] = numbers[j + 1], numbers[j]
                # Set the flag to True to indicate that a swap was made
                swapped = True

        # If no swaps were made in the current iteration, the list is already sorted
        if not swapped:
            break

    return numbers
